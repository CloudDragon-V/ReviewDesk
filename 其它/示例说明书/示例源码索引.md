# 两份说明书对应的完整示例源码

以下均为本项目合成的教学/测评源码。生成说明时只静态解析，没有执行。

## 购物订单（三文件）

### main.py

```python
"""入口模块：读取购物项，计算订单，输出用户可读的账单。"""
from pricing import calculate_order
from presentation import format_receipt


def main():
    """准备输入，再依次调用计算模块和展示模块。"""
    items = [{"name": "笔记本", "price": 12.5, "quantity": 2},
             {"name": "水笔", "price": 3.0, "quantity": 4}]
    member = True
    order = calculate_order(items, member)
    receipt = format_receipt(order)
    print(receipt)
    return order


if __name__ == "__main__":
    main()
```

### pricing.py

```python
"""计价模块：验证单价与数量，累计金额，按会员规则折扣。"""


def validate_item(item):
    """要求单价非负、数量为正，否则抛出异常。"""
    if item["price"] < 0:
        raise ValueError("单价不能为负数")
    if item["quantity"] <= 0:
        raise ValueError("数量必须大于零")
    return item["price"] * item["quantity"]


def calculate_order(items, member=False):
    """接收商品列表，返回原价、折扣额、实付额与逐项金额。"""
    subtotal = 0.0
    details = []
    for item in items:
        amount = validate_item(item)
        subtotal += amount
        details.append({"name": item["name"], "amount": amount})
    discount = 0.0
    if member and subtotal >= 30:
        discount = subtotal * 0.1
    total = round(subtotal - discount, 2)
    return {"subtotal": subtotal, "discount": discount,
            "total": total, "details": details}
```

### presentation.py

```python
"""展示模块：将订单字典转成便于阅读的多行文本。"""


def format_receipt(order):
    """读取金额字段，组织清单，返回文本；不重新计算计价规则。"""
    lines = ["订单清单"]
    for detail in order["details"]:
        lines.append(f"{detail['name']}：{detail['amount']:.2f} 元")
    lines.append(f"原价：{order['subtotal']:.2f} 元")
    lines.append(f"优惠：{order['discount']:.2f} 元")
    lines.append(f"实付：{order['total']:.2f} 元")
    return "\n".join(lines)
```

## 协程调度（两文件）

### main.py

```python
import asyncio
from worker import assemble, summarize


def main():
    rows, before, log = asyncio.run(assemble())
    print(summarize(rows))
    print(before)
    print(log)
    return rows, before, log


if __name__ == "__main__":
    main()
```

### worker.py

```python
import asyncio


async def transform(name, amount, log):
    log.append(f"start:{name}")
    await asyncio.sleep(0)
    log.append(f"finish:{name}")
    return {"name": name, "amount": amount * 2}


async def assemble():
    log = []
    pending = transform("A", 3, log)
    before = list(log)
    second = await transform("B", 2, log)
    first = await pending
    return [first, second], before, log


def summarize(rows):
    return sum(row["amount"] for row in rows if row["amount"] >= 5)
```
